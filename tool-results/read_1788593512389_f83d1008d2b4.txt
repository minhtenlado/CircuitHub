   156→interface SearchResult {
   157→  products: Array<{
   158→    id: string;
   159→    name: string;
   160→    slug: string;
   161→    price: number;
   162→    productType: string;
   163→    brand: string | null;
   164→    rating: number;
   165→    imageUrl?: string;
   166→    shopName?: string;
   167→    shopVerified?: boolean;
   168→  }>;
   169→  categories: Array<{ id: string; name: string; slug: string; icon?: string | null }>;
   170→  shops: Array<{
   171→    id: string;
   172→    name: string;
   173→    slug: string;
   174→    logoUrl?: string | null;
   175→    verified: boolean;
   176→    rating: number;
   177→  }>;
   178→  brands: string[];
   179→}
   180→
   181→function SearchBar({ compact = false }: { compact?: boolean }) {
   182→  const goProducts = useNavStore((s) => s.goProducts);
   183→  const goProduct = useNavStore((s) => s.goProduct);
   184→  const goCategory = useNavStore((s) => s.goCategory);
   185→  const goShop = useNavStore((s) => s.goShop);
   186→  const searchHistory = useSearchHistoryStore();
   187→  const [query, setQuery] = useState('');
   188→  const [focused, setFocused] = useState(false);
   189→  const [results, setResults] = useState<SearchResult | null>(null);
   190→  const [loading, setLoading] = useState(false);
   191→  const [activeIndex, setActiveIndex] = useState(-1);
   192→  const wrapperRef = useRef<HTMLDivElement>(null);
   193→  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   194→
   195→  // Close dropdown on outside click
   196→  useEffect(() => {
   197→    const onClick = (e: MouseEvent) => {
   198→      if (!wrapperRef.current?.contains(e.target as Node)) setFocused(false);
   199→    };
   200→    document.addEventListener('mousedown', onClick);
   201→    return () => document.removeEventListener('mousedown', onClick);
   202→  }, []);
   203→
   204→  // Debounced search
   205→  useEffect(() => {
   206→    if (debounceRef.current) clearTimeout(debounceRef.current);
   207→    const q = query.trim();
   208→    if (q.length < 2) {
   209→      setResults(null);
   210→      setLoading(false);
   211→      return;
   212→    }
   213→    setLoading(true);
   214→    debounceRef.current = setTimeout(async () => {
   215→      try {
   216→        const res = await fetch(`/api/v1/search?q=${encodeURIComponent(q)}`);
   217→        const json = await res.json();
   218→        if (json.success) {
   219→          setResults(json.data);
   220→        }
   221→      } catch {
   222→        // silently fail
   223→      } finally {
   224→        setLoading(false);
   225→      }
   226→    }, 250);
   227→    return () => {
   228→      if (debounceRef.current) clearTimeout(debounceRef.current);
   229→    };
   230→  }, [query]);
   231→
   232→  const submit = (q?: string) => {
   233→    const term = (q ?? query).trim();
   234→    if (term) searchHistory.add(term);
   235→    goProducts(term ? { q: term } : {});
   236→    setFocused(false);
   237→  };
   238→
   239→  const hasResults = results && (
   240→    results.products.length > 0 ||
   241→    results.categories.length > 0 ||
   242→    results.shops.length > 0 ||
   243→    results.brands.length > 0
   244→  );
   245→
   246→  // Flatten all items for keyboard navigation
   247→  const flatItems: Array<{ type: 'product' | 'category' | 'shop' | 'brand'; data: any; label: string }> = [];
   248→  if (results) {
   249→    results.products.forEach((p) => flatItems.push({ type: 'product', data: p, label: p.name }));
   250→    results.categories.forEach((c) => flatItems.push({ type: 'category', data: c, label: c.name }));
   251→    results.shops.forEach((s) => flatItems.push({ type: 'shop', data: s, label: s.name }));
   252→    results.brands.forEach((b) => flatItems.push({ type: 'brand', data: b, label: b }));
   253→  }
   254→
   255→  function handleKeyDown(e: React.KeyboardEvent) {
   256→    if (e.key === 'Enter') {
   257→      if (activeIndex >= 0 && activeIndex < flatItems.length) {
   258→        const item = flatItems[activeIndex];
   259→        handleSelect(item.type, item.data);
   260→      } else {
   261→        submit();
   262→      }
   263→      return;
   264→    }
   265→    if (e.key === 'ArrowDown') {
   266→      e.preventDefault();
   267→      setActiveIndex((i) => Math.min(i + 1, flatItems.length - 1));
   268→    } else if (e.key === 'ArrowUp') {
   269→      e.preventDefault();
   270→      setActiveIndex((i) => Math.max(i - 1, -1));
   271→    } else if (e.key === 'Escape') {
   272→      setFocused(false);
   273→    }
   274→  }
   275→
   276→  function handleSelect(type: string, data: any) {
   277→    setFocused(false);
   278→    if (type === 'product') goProduct(data.slug);
   279→    else if (type === 'category') goCategory(data.slug);
   280→    else if (type === 'shop') goShop(data.slug);
   281→    else if (type === 'brand') goProducts({ q: data });
   282→  }
   283→
   284→  return (
   285→    <div ref={wrapperRef} className="relative w-full">
   286→      <div className="relative">
   287→        <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   288→        <Input
   289→          value={query}
   290→          onChange={(e) => {
   291→            setQuery(e.target.value);
   292→            setActiveIndex(-1);
   293→          }}
   294→          onFocus={() => setFocused(true)}
   295→          onKeyDown={handleKeyDown}
   296→          placeholder="Search by product, MPN, KiCad project, component, service..."
   297→          aria-label="Search CircuitHub"
   298→          className={cn(
   299→            'h-10 rounded-full border-border/70 bg-background/70 pl-9 pr-9 text-sm shadow-xs transition-all',
   300→            'focus-visible:border-primary focus-visible:ring-primary/30 focus-visible:ring-[3px]',
   301→            compact && 'h-9',
   302→          )}
   303→        />
   304→        {loading && (
   305→          <div className="absolute right-3 top-1/2 -translate-y-1/2">
   306→            <div className="h-3.5 w-3.5 animate-spin rounded-full border-2 border-cyan-300 border-t-cyan-600" />
   307→          </div>
   308→        )}
   309→        {!loading && query && (
   310→          <button
   311→            onClick={() => { setQuery(''); setResults(null); }}
   312→            className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   313→            aria-label="Clear search"
   314→          >
   315→            <svg className="h-3.5 w-3.5" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">
   316→              <path d="M18 6L6 18M6 6l12 12" />
   317→            </svg>
   318→          </button>
   319→        )}
   320→      </div>
   321→
   322→      {/* Autocomplete dropdown */}
   323→      <AnimatePresence>
   324→        {focused && (
   325→          <motion.div
   326→            initial={{ opacity: 0, y: -4 }}
   327→            animate={{ opacity: 1, y: 0 }}
   328→            exit={{ opacity: 0, y: -4 }}
   329→            transition={{ duration: 0.15 }}
   330→            className="absolute left-0 right-0 top-[calc(100%+6px)] z-50 rounded-xl border border-border/70 bg-popover/95 shadow-lg backdrop-blur-md max-h-[70vh] overflow-y-auto"
   331→          >
   332→            {/* No query: show recent + popular searches */}
   333→            {!query.trim() && (
   334→              <div className="p-3 space-y-3">
   335→                {searchHistory.searches.length > 0 && (
   336→                  <div>
   337→                    <div className="flex items-center justify-between px-1 pb-1.5">
   338→                      <span className="text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   339→                        Recent searches
   340→                      </span>
   341→                      <button
   342→                        onClick={() => searchHistory.clear()}
   343→                        className="text-[10px] text-muted-foreground hover:text-rose-500 transition-colors"
   344→                      >
   345→                        Clear
   346→                      </button>
   347→                    </div>
   348→                    <div className="space-y-0.5">
   349→                      {searchHistory.searches.map((s) => (
   350→                        <button
   351→                          key={s}
   352→                          onClick={() => { setQuery(s); submit(s); }}
   353→                          className="flex items-center gap-2 w-full rounded-md px-2 py-1.5 text-sm text-left hover:bg-cyan-50 transition-colors group"
   354→                        >
   355→                          <Clock className="h-3.5 w-3.5 text-muted-foreground group-hover:text-cyan-600 flex-shrink-0" />
   356→                          <span className="flex-1 truncate text-slate-700">{s}</span>
   357→                          <span
   358→                            onClick={(e) => { e.stopPropagation(); searchHistory.remove(s); }}
   359→                            className="opacity-0 group-hover:opacity-100 text-muted-foreground hover:text-rose-500 p-0.5 rounded transition-all"
   360→                          >
   361→                            <X className="h-3 w-3" />
   362→                          </span>
   363→                        </button>
   364→                      ))}
   365→                    </div>
   366→                  </div>
   367→                )}
   368→                <div>
   369→                  <div className="px-1 pb-1.5 text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   370→                    Popular searches
   371→                  </div>
   372→                  <div className="flex flex-wrap gap-1.5">
   373→                    {POPULAR_SEARCHES.map((s) => (
   374→                      <button
   375→                        key={s}
   376→                        onClick={() => { setQuery(s); submit(s); }}
   377→                        className="rounded-full border border-cyan-200/60 bg-cyan-50/60 px-2.5 py-1 text-xs font-medium text-cyan-800 transition-colors hover:border-primary hover:bg-primary hover:text-primary-foreground"
   378→                      >
   379→                        {s}
   380→                      </button>
   381→                    ))}
   382→                  </div>
   383→                </div>
   384→              </div>
   385→            )}
   386→
   387→            {/* Query too short */}
   388→            {query.trim().length > 0 && query.trim().length < 2 && (
   389→              <div className="p-4 text-center text-sm text-muted-foreground">
   390→                Keep typing... (min 2 characters)
   391→              </div>
   392→            )}
   393→
   394→            {/* Loading */}
   395→            {query.trim().length >= 2 && loading && (
   396→              <div className="p-4 text-center text-sm text-muted-foreground flex items-center justify-center gap-2">
   397→                <div className="h-3.5 w-3.5 animate-spin rounded-full border-2 border-cyan-300 border-t-cyan-600" />
   398→                Searching...
   399→              </div>
   400→            )}
   401→
   402→            {/* Results */}
   403→            {query.trim().length >= 2 && !loading && hasResults && (
   404→              <div className="p-2">
   405→                {/* Products */}
   406→                {results!.products.length > 0 && (
   407→                  <div className="mb-2">
   408→                    <div className="px-2 py-1 text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   409→                      Products ({results!.products.length})
   410→                    </div>
   411→                    {results!.products.map((p, i) => {

   413→                      return (
   414→                        <button
   415→                          key={p.id}
   416→                          onClick={() => handleSelect('product', p)}
   417→                          onMouseEnter={() => setActiveIndex(idx)}
   418→                          className={cn(
   419→                            'flex w-full items-center gap-2.5 rounded-lg p-2 text-left transition-colors',
   420→                            activeIndex === idx ? 'bg-cyan-50' : 'hover:bg-slate-50',
   421→                          )}
   422→                        >
   423→                          <div className="h-9 w-9 flex-shrink-0 overflow-hidden rounded-md bg-muted border border-border/40">
   424→                            {p.imageUrl && (
   425→                              <img src={p.imageUrl} alt={p.name} className="h-full w-full object-cover" />
   426→                            )}
   427→                          </div>
   428→                          <div className="flex-1 min-w-0">
   429→                            <p className="text-sm font-medium truncate">{p.name}</p>
   430→                            <p className="text-xs text-muted-foreground">
   431→                              {p.brand ? `${p.brand} · ` : ''}{p.shopName ?? 'Shop'}
   432→                            </p>
   433→                          </div>
   434→                          <span className="text-xs font-bold text-cyan-700 tabular-nums flex-shrink-0">
   435→                            {new Intl.NumberFormat('vi-VN').format(p.price)}₫
   436→                          </span>
   437→                        </button>
   438→                      );
   439→                    })}
   440→                  </div>
   441→                )}
   442→
   443→                {/* Categories */}
   444→                {results!.categories.length > 0 && (
   445→                  <div className="mb-2">
   446→                    <div className="px-2 py-1 text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   447→                      Categories
   448→                    </div>
   449→                    {results!.categories.map((c) => {

   451→                      return (
   452→                        <button
   453→                          key={c.id}
   454→                          onClick={() => handleSelect('category', c)}
   455→                          onMouseEnter={() => setActiveIndex(idx)}
   456→                          className={cn(
   457→                            'flex w-full items-center gap-2.5 rounded-lg p-2 text-left transition-colors',
   458→                            activeIndex === idx ? 'bg-cyan-50' : 'hover:bg-slate-50',
   459→                          )}
   460→                        >
   461→                          <div className="h-8 w-8 flex-shrink-0 rounded-md bg-cyan-50 border border-cyan-100 flex items-center justify-center">

   467→                      );
   468→                    })}
   469→                  </div>
   470→                )}
   471→
   472→                {/* Shops */}
   473→                {results!.shops.length > 0 && (

   475→                    <div className="px-2 py-1 text-[11px] font-medium uppercase tracking-wider text-muted-foreground">
   476→                      Shops
   477→                    </div>
   478→                    {results!.shops.map((s) => {

   480→                      return (
   481→                        <button
   482→                          key={s.id}
   483→                          onClick={() => handleSelect('shop', s)}
   484→                          onMouseEnter={() => setActiveIndex(idx)}
   485→                          className={cn(
   486→                            'flex w-full items-center gap-2.5 rounded-lg p-2 text-left transition-colors',
   487→                            activeIndex === idx ? 'bg-cyan-50' : 'hover:bg-slate-50',

   489→                        >
