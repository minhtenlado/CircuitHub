   708→/* ---------------- Order timeline ---------------- */
   709→
   710→function OrderTimeline({ status, createdAt }: { status: string; createdAt: string }) {
   711→  const completedSteps = STATUS_TO_STEPS[status] ?? 1;
   712→  const base = new Date(createdAt).getTime();
   713→  const isCancelled = status === 'CANCELLED';
   714→  const steps = TIMELINE_STEPS.map((s, idx) => {
   715→    const isComplete = idx < completedSteps;
   716→    const ts = new Date(base + idx * 6 * 3600000).toISOString();
   717→    return { ...s, isComplete, ts };
   718→  });
   719→
   720→  return (
   721→    <ol className="relative space-y-4 pl-6">
   722→      {steps.map((s, idx) => {
   723→        const isLast = idx === steps.length - 1;
   724→        return (
   725→          <li key={s.id} className="relative">
   726→            {/* Connector */}
   727→            {!isLast && (
   728→              <span
   729→                aria-hidden
   730→                className={cn(
   731→                  'absolute left-[7px] top-5 bottom-[-1rem] w-px',
   732→                  s.isComplete ? 'bg-cyan-300' : 'bg-border',
   733→                )}
   734→              />
   735→            )}
   736→            {/* Dot */}
   737→            <span
   738→              className={cn(
   739→                'absolute -left-6 top-0.5 flex h-3.5 w-3.5 items-center justify-center rounded-full border-2',
   740→                isCancelled && idx === 0 ? 'border-red-400 bg-red-500' : s.isComplete ? 'border-cyan-500 bg-cyan-500' : 'border-border bg-background',
   741→              )}
   742→            >
   743→              {s.isComplete && !isCancelled && <CheckCircle2 className="h-2.5 w-2.5 text-white" />}
   744→            </span>
   745→            {/* Label + time */}
   746→            <div className="flex items-baseline justify-between gap-3">
   747→              <p className={cn('text-sm font-medium', s.isComplete ? 'text-foreground' : 'text-muted-foreground')}>
   748→                {s.label}
   749→                {isCancelled && idx === 0 && <span className="ml-1.5 text-red-600">· cancelled</span>}
   750→              </p>
   751→              <time className="text-xs text-muted-foreground tabular-nums whitespace-nowrap">
   752→                {s.isComplete ? formatDate(s.ts, { hour: '2-digit', minute: '2-digit' }) : '—'}
   753→              </time>
   754→            </div>
   755→          </li>
   756→        );
   757→      })}
   758→    </ol>
   759→  );
   760→}
